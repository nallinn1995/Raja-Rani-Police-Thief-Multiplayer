export type ConnectionMode = "tunnel" | "localhost";

const MODE_KEY = "app_connection_mode";

class ConfigService {
  private mode: ConnectionMode;
  private listeners: Array<(mode: ConnectionMode) => void> = [];

  constructor() {
    const saved = typeof window !== "undefined" ? (localStorage.getItem(MODE_KEY) as ConnectionMode) : null;
    if (saved === "localhost" || saved === "tunnel") {
      this.mode = saved;
    } else {
      // Default to tunnel (auto relative mode, which works through Vite proxy & Cloudflare tunnels)
      this.mode = "tunnel";
    }
  }

  public getMode(): ConnectionMode {
    return this.mode;
  }

  public setMode(mode: ConnectionMode): void {
    if (this.mode !== mode) {
      this.mode = mode;
      if (typeof window !== "undefined") {
        localStorage.setItem(MODE_KEY, mode);
      }
      this.listeners.forEach((cb) => cb(mode));
    }
  }

  public subscribe(cb: (mode: ConnectionMode) => void): () => void {
    this.listeners.push(cb);
    return () => {
      this.listeners = this.listeners.filter((l) => l !== cb);
    };
  }

  public getApiBaseUrl(): string {
    if (this.mode === "localhost") {
      return "http://localhost:3001";
    }
    return import.meta.env.VITE_API_BASE_URL || "";
  }

  public getSocketUrl(): string | undefined {
    if (this.mode === "localhost") {
      return "http://localhost:3001";
    }
    return import.meta.env.VITE_SERVER_URL || undefined;
  }
}

export const configService = new ConfigService();
